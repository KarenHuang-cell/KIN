                if(e.isIntersecting){
                    wrap.classList.add('kin-mm-wrap--shown');

                }
            });
        },{threshold:.2});

  3. Select all the existing code (⌘A inside the embed dialog) and delete it.
  4. Paste the new code from the dark box above into the embed dialog.
  5. Click Save & Close on the embed dialog.

  8. Save the CMS item, then Publish the site.
